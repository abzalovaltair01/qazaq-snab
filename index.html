async function calc() { const txt = document.getElementById("txt").value; const btn = document.getElementById("btn"); if(!txt) return alert("Введите список"); btn.disabled = true; btn.innerText = "РАБОТАЕМ..."; try { const res = await fetch("/api/calculate", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ materials: txt }) }); const data = await res.json(); // Проверяем, есть ли данные в ответе if (data.error) throw new Error(data.error); if (!data.choices) throw new Error("Сервер вернул пустой ответ"); const content = data.choices[0].message.content; const match = content.match(/\[.*\]/s); if (!match) throw new Error("ИИ прислал текст вместо таблицы"); const items = JSON.parse(match[0]); let sum = 0; let html = ""; items.forEach(i => { sum += i.p; html += `${i.n}${i.p.toLocaleString()} ₸`; }); document.getElementById("table").innerHTML = html; document.getElementById("total").innerText = "ИТОГО: " + sum.toLocaleString() + " ₸"; document.getElementById("result").style.display = "block"; } catch (e) { alert("Ошибка: " + e.message); console.log("Полный ответ сервера:", e); } finally { btn.disabled = false; btn.innerText = "РАССЧИТАТЬ"; } }
